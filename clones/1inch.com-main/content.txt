        const VERSION_CHECK_INTERVAL_MS = 60000; // 1 minute
        const HARD_RELOAD_DELAY_MS = 3000; // 3 seconds
        const FETCH_TIMEOUT_MS = 5000; // 5 seconds

        const log = (message) => console.debug('[1inch]', message);

        // Returns: Array of 3 numbers [major, minor, patch] or null if invalid
        const parseVersion = (v) => {
          if (!v || typeof v !== 'string') return null;
          const parts = v.split('.').map((p) => parseInt(p, 10));
          if (parts.length !== 3 || parts.some(isNaN)) return null;
          return parts;
        };

        // Returns: -1 if a < b, 1 if a > b, 0 if a === b
        const compareVersionParts = (a, b) => {
          for (let i = 0; i < 3; i++) {
            if (a[i] < b[i]) return -1;
            if (a[i] > b[i]) return 1;
          }
          return 0;
        };

        // Returns: {status: 'Update required'|'Up to date'|'Unable to compare'|'Comparison failed', color: string}
        const compareVersions = (current, minHard) => {
          try {
            const currentVersionNumbers = parseVersion(current);
            const minHardVersionNumbers = parseVersion(minHard);

            if (!currentVersionNumbers || !minHardVersionNumbers) {
              return { status: 'Unable to compare', color: 'color: #9ca3af' };
            }

            if (compareVersionParts(currentVersionNumbers, minHardVersionNumbers) < 0) {
              return { status: 'Update required', color: 'color: #ef4444; font-weight: bold' };
            }

            return { status: 'Up to date', color: 'color: #10b981; font-weight: bold' };
          } catch (e) {
            return { status: 'Comparison failed', color: 'color: #9ca3af' };
          }
        };

        const hardReload = (currentVersion, targetVersion) => {
          log('Reloading in 3 seconds to update from ' + currentVersion + ' → ' + targetVersion);
          window.__reloadPending = true;
          try {
            sessionStorage.setItem('1inch:reloaded', '1');
          } catch (e) {}
          setTimeout(() => {
            window.location.reload(true);
          }, HARD_RELOAD_DELAY_MS);
        };

        const checkRemoteVersion = (currentVersion) => {

            xhr.timeout = FETCH_TIMEOUT_MS;

            xhr.onload = () => {
              if (xhr.status !== 200) return;

              const contentType = xhr.getResponseHeader('content-type');
              if (!contentType || contentType.indexOf('application/json') === -1) return;

              let data;
              try {
                data = JSON.parse(xhr.responseText);
              } catch (e) {
                return;
              }

              }
            };

            xhr.onerror = xhr.ontimeout = () => {};

            xhr.send();
          } catch (e) {

        };
